+package main

+import (
+	"log"

+	"bifrost-proxy/api"
+	"bifrost-proxy/engine"
+	"bifrost-proxy/storage"
+

+
+func main() {
+	app := fiber.New(fiber.Config{
+		// Optimize for SLA
+		DisableStartupMessage: true,
+	})
+
+	// 1. Storage setup
+	endpoint := os.Getenv("S3_ENDPOINT")
+	if endpoint == "" {
+		endpoint = "localhost:9000"
+	}
+	
+	accessKey := os.Getenv("AWS_ACCESS_KEY_ID")
+	if accessKey == "" {
+		accessKey = "admin"
+	}
+	
+	secretKey := os.Getenv("AWS_SECRET_ACCESS_KEY")
+	if secretKey == "" {
+		secretKey = "password"
+	}
+
+	bucket := os.Getenv("S3_BUCKET")
+	if bucket == "" {
+		bucket = "lakehouse"
+	}
+
+	store, err := storage.NewMinioStorage(endpoint, accessKey, secretKey, bucket)
+	if err != nil {
+		log.Printf("Warning: Failed to init real S3 storage: %v", err)
+	}
+
+	// 2. DuckDB Engine setup
+	syncEngine, err := engine.NewSyncEngine()
+	if err != nil {
+		log.Printf("Warning: Failed to init DuckDB engine: %v", err)
+	} else {
+		defer syncEngine.Close()
+	}
+
+	// 3. Mount Routes
+	api.SetupAuthRoutes(app)
+	api.SetupCatalogRoutes(app, store, syncEngine)
+
+	log.Println("Bifrost Proxy is starting on :8080...")
+	log.Fatal(app.Listen(":8080"))
+}

+import (
+	"context"

+
+var ErrTableNotFound = errors.New("table not found")
+
+// StorageService defines the interface for interacting with underlying storage
+type StorageService interface {
+	GetTableMetadata(ctx context.Context, namespace, table string) ([]byte, error)
+	SaveTableMetadata(ctx context.Context, namespace, table string, data []byte) error
+}
+
+type MinioStorage struct {
+	client *minio.Client
+	bucket string
+}
+
+func NewMinioStorage(endpoint, accessKey, secretKey, bucket string) (*MinioStorage, error) {
+	client, err := minio.New(endpoint, &minio.Options{
+		Creds:  credentials.NewStaticV4(accessKey, secretKey, ""),
+		Secure: false,
+	})
+	if err != nil {
+		return nil, err
+	}
+	return &MinioStorage{
+		client: client,
+		bucket: bucket,
+	}, nil
+}
+
+func (s *MinioStorage) GetTableMetadata(ctx context.Context, namespace, table string) ([]byte, error) {
+	objectName := fmt.Sprintf("%s/%s/metadata.json", namespace, table)
+	
+	obj, err := s.client.GetObject(ctx, s.bucket, objectName, minio.GetObjectOptions{})
+	if err != nil {
+		return nil, err
+	}
+	defer obj.Close()
+
+	_, err = obj.Stat()
+	if err != nil {
+		errResp := minio.ToErrorResponse(err)
+		if errResp.Code == "NoSuchKey" {
+			return nil, ErrTableNotFound
+		}
+		return nil, err
+	}
+
+	data, err := io.ReadAll(obj)
+	if err != nil {
+		return nil, err
+	}
+	return data, nil
+}
